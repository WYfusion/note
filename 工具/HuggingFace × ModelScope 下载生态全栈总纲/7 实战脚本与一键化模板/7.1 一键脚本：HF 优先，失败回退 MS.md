> 这是本仓库最出名的「万能脚本」：不管你是 Windows / Linux / macOS，HF 镜像走不通就自动回退到 ModelScope。三种调用方式：**bash 单文件 / Python 模块 / Makefile 任务**。

## bash 脚本（推荐）

```bash
#!/usr/bin/env bash
# get-model.sh——HF 镜像优先，失败后 MS
# 用法：get-model.sh <hf_repo> [<ms_repo>] [<out_dir>]
set -euo pipefail

HF_REPO="$1"
MS_REPO="${2:-$HF_REPO}"           # 不指定则同名
OUT="${3:-./models/$(basename "$HF_REPO")}"

echo "[1/2] Trying HF mirror..."
if HF_ENDPOINT=https://hf-mirror.com HF_HUB_ENABLE_HF_TRANSFER=1 \
   huggingface-cli download "$HF_REPO" --local-dir "$OUT" --local-dir-use-symlinks False 2>/tmp/hf.err; then
  echo "✓ HF mirror OK"
  exit 0
fi
echo "✗ HF mirror failed:"
tail -20 /tmp/hf.err

echo "[2/2] Falling back to ModelScope..."
python - <<PY
from modelscope import snapshot_download
import sys
p = snapshot_download("${MS_REPO}", cache_dir="${OUT}.ms")
print("Downloaded to:", p)
import os, shutil
if os.path.exists("${OUT}"):
    shutil.rmtree("${OUT}")
shutil.copytree(p, "${OUT}")
print("Linked at:", "${OUT}")
PY
```

跑：

```bash
chmod +x get-model.sh
./get-model.sh Qwen/Qwen2.5-7B-Instruct qwen/Qwen2.5-7B-Instruct
```

## Python 模块版本

```python
# get_model.py
import os, shutil, traceback
from pathlib import Path

def get_model(hf_repo, ms_repo=None, out=None):
    ms_repo = ms_repo or hf_repo
    out = Path(out or f"./models/{Path(hf_repo).name}")
    out.parent.mkdir(parents=True, exist_ok=True)

    # 1. HF mirror
    try:
        os.environ.setdefault("HF_ENDPOINT", "https://hf-mirror.com")
        os.environ.setdefault("HF_HUB_ENABLE_HF_TRANSFER", "1")
        from huggingface_hub import snapshot_download
        local = snapshot_download(hf_repo, local_dir=str(out), local_dir_use_symlinks=False)
        print("✓ HF:", local); return local
    except Exception as e:
        traceback.print_exc()
        print("✗ HF failed:", e)

    # 2. ModelScope
    from modelscope import snapshot_download as ms_dl
    local = ms_dl(ms_repo, cache_dir=str(out) + ".ms")
    if out.exists(): shutil.rmtree(out)
    shutil.copytree(local, out)
    print("✓ MS:", out); return str(out)

if __name__ == "__main__":
    import sys
    get_model(*sys.argv[1:])
```

## Makefile 集成

```makefile
QWEN := Qwen/Qwen2.5-7B-Instruct
MODELS_DIR := ./models

.PHONY: qwen
qwen:
	./get-model.sh $(QWEN) qwen/Qwen2.5-7B-Instruct $(MODELS_DIR)/qwen2.5-7b

.PHONY: clean-models
clean-models:
	rm -rf $(MODELS_DIR)
```

## 错误分发与重试

- 401/403：仓库应是 Gated，切到官站 + token 重试。脚本中加一个分叉判断。

- 429：限流，插入10–60s sleep 重试。

- HTTP 5xx：镜像临时劳总部问题，重新跑一遍或额外增加 hfd 回退。

- 三阶回退：hf-mirror → [huggingface.co](http://huggingface.co) → ModelScope。

## 增强点 idea
